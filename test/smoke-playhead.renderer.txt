(async()=>{
  const source=await window.scoresprout.importMidi();
  showSource(source);
  const track=source.tracks.find(t=>t.detected==='piano')||source.tracks[0];
  const generated=await window.scoresprout.generate(source.id,track.id,track.detected,'full',source.title,false);
  await renderVersion(generated);
  if(!scorePlayhead||scoreAnchors.length<2)throw Error('谱面光标没有关联音符');
  const pair=scoreAnchors.findIndex((anchor,index)=>{
    const next=scoreAnchors[index+1];
    return next&&anchor.page===next.page&&Math.abs(anchor.y-next.y)<4&&next.tick>anchor.tick;
  });
  if(pair<0)throw Error('没有可供连续移动的相邻音符');
  const first=scoreAnchors[pair],second=scoreAnchors[pair+1];
  const midpoint=(first.tick+second.tick)/2*60/(generated.version.bpm*generated.version.ppq);
  paintScoreCursor(midpoint);
  const actual=scorePlayhead.getBoundingClientRect().left-first.page.getBoundingClientRect().left+1;
  if(Math.abs(actual-(first.x+second.x)/2)>1.5)throw Error('竖线没有在相邻音符之间平滑插值');
  const wrap=scoreAnchors.findIndex((anchor,index)=>scoreAnchors[index+1]&&
    (anchor.page!==scoreAnchors[index+1].page||Math.abs(anchor.y-scoreAnchors[index+1].y)>=4));
  if(wrap>=0){
    const next=scoreAnchors[wrap+1];
    paintScoreCursor(next.tick*60/(generated.version.bpm*generated.version.ppq));
    const y=scorePlayhead.getBoundingClientRect().top-next.page.getBoundingClientRect().top;
    if(Math.abs(y-next.y)>1.5)throw Error('换行后竖线没有切到新谱表');
  }
  await player.play();
  await new Promise(resolve=>setTimeout(resolve,650));
  const rect=scorePlayhead.getBoundingClientRect();
  if(!player.playing||!scorePlayhead.isConnected||
    // Windows fractional DPI may report a 2 CSS px line as 1.994 px.
    scorePlayhead.style.display!=='block'||rect.width<1.5||rect.height<50)
    throw Error(`播放时谱面竖线不可见：${JSON.stringify({playing:player.playing,connected:scorePlayhead.isConnected,
      display:scorePlayhead.style.display,width:rect.width,height:rect.height,style:scorePlayhead.getAttribute('style')})}`);
  player.pause();player.seek(2);
  if(scoreAnchorIndex===0)throw Error('进度跳转没有移动谱面竖线');
  const seekIndex=scoreAnchorIndex;
  player.stop();
  if(scorePlayhead.style.display!=='none')throw Error('停止后竖线没有隐藏');
  const oldWidth=$('score').clientWidth;
  $('score').style.width='70%';
  await new Promise(resolve=>setTimeout(resolve,450));
  if($('score').clientWidth>=oldWidth||!scoreAnchors.length)throw Error('缩窄窗口后谱面没有重新排版');
  const anchor=scoreAnchors[Math.min(2,scoreAnchors.length-1)];
  paintScoreCursor(anchor.tick*60/(generated.version.bpm*generated.version.ppq));
  const resized=scorePlayhead.getBoundingClientRect();
  if(!scorePlayhead.isConnected||Math.abs(resized.left-anchor.page.getBoundingClientRect().left+1-anchor.x)>1.5)
    throw Error('重新排版后竖线没有对齐音符');
  $('score').style.width='';
  await new Promise(resolve=>setTimeout(resolve,450));
  await player.play();
  return {visible:true,interpolated:true,resized:true,wrapped:wrap>=0,seekIndex,playing:player.playing,notes:generated.version.notes.length};
})()
