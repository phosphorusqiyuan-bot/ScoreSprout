(async()=>{
  const check=(condition,message)=>{if(!condition)throw Error(message);};
  const items=await window.scoresprout.listLibrary(),item=items[0];
  check(item,'排版检查需要一个已保存版本');
  const loaded=await window.scoresprout.loadVersion(item.songId,item.id);
  $('workspace').classList.remove('hidden');$('emptyState').classList.add('hidden');
  await renderVersion(loaded);
  check(new DOMParser().parseFromString(loaded.musicxml,'application/xml').querySelector('parsererror')===null,'MusicXML 格式无效');
  const notes=JSON.stringify(loaded.version.notes),wanted=new Set(loaded.version.notes.map(n=>n.start));
  check(scoreAnchors.length===wanted.size,'排版后光标起音丢失');
  const beforeWidth=$('score').querySelector('svg').getBoundingClientRect().width;
  $('scoreZoom').value='115';redrawScore();
  check(scoreAnchors.length===wanted.size,'缩放后光标起音丢失');
  $('showFingerings').checked=false;redrawScore();
  check(!scoreDisplay.EngravingRules.RenderFingerings,'隐藏指法未生效');
  check(!scoreDisplay.GraphicSheet.MeasureList.flat().some(m=>m.staffEntries.some(s=>s.FingeringEntries.length)),'隐藏后仍绘制旧指法');
  check(JSON.stringify(loaded.version.notes)===notes,'排版改动了试听音符');
  $('showFingerings').checked=true;$('scoreZoom').value='100';redrawScore();
  player.seek(Math.min(1,player.duration/2));
  check(scorePlayhead.style.display==='block','排版后进度光标不可见');
  const exported=await window.scoresprout.exportVersion(item.songId,item.id,'musicxml');
  check(exported,'排版后的 MusicXML 未导出');
  const report={title:loaded.version.title,notes:loaded.version.notes.length,anchors:scoreAnchors.length,
    beams:(loaded.musicxml.match(/<beam\b/g)||[]).length,zoom:true,fingeringToggle:true,exported,beforeWidth,
    fingeringLabels:scoreDisplay.GraphicSheet.MeasureList.flat().flatMap(m=>m.staffEntries).reduce((sum,s)=>sum+s.FingeringEntries.length,0),
    fingeringModifiers:scoreDisplay.GraphicSheet.MeasureList.flat().flatMap(m=>m.staffEntries)
      .flatMap(s=>s.graphicalVoiceEntries).flatMap(v=>v.vfStaveNote?.getModifiers?.()||[])
      .filter(m=>/stringnumber|frethandfinger/i.test(m.getCategory?.()||'')).length};
  check(report.fingeringModifiers===0,'仍有重复的 VexFlow 指法数字');
  check(report.fingeringLabels>0||!loaded.version.pianoProcessing?.fingeringCount,'指法重新显示未生效');
  player.stop();clearTimeout(scoreResizeTimer);scoreDisplay=null;
  document.body.innerHTML=$('score').querySelector('svg').outerHTML;document.body.style.background='white';
  window.scrollTo(0,0);
  await new Promise(resolve=>requestAnimationFrame(()=>requestAnimationFrame(resolve)));
  return report;
})()
