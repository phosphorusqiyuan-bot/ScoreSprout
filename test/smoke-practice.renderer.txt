(async()=>{
  const check=(value,message)=>{if(!value)throw Error(message);};
  const items=await window.scoresprout.listLibrary(),item=items[0];check(item,'No saved score');
  const data=await window.scoresprout.loadVersion(item.songId,item.id),original=JSON.stringify(data.version.notes);
  $('workspace').classList.remove('hidden');$('emptyState').classList.add('hidden');
  await renderVersion(data);
  const tiny=xml=>(xml.match(/<type>(?:32nd|64th)<\/type>/g)||[]).length;
  check(!document.getElementById('readableScore'),'Readable view toggle should be absent');
  check(notationVersion.readability,'Readable score not loaded');
  const before=tiny(data.musicxml),after=tiny(data.readable.musicxml);
  check(after<=before,'Readability adds tiny notes');
  check(scoreAnchors.length===new Set(notationVersion.notes.map(n=>n.start)).size,'Readable attacks missing from cursor');
  check(scoreAnchors.every((a,i)=>!i||a.tick>=scoreAnchors[i-1].tick),'Performance cursor moves backwards');
  $('showFingerings').checked=false;redrawScore();$('showFingerings').checked=true;redrawScore();
  const fingers=scoreDisplay.GraphicSheet.MeasureList.flat().filter(Boolean).flatMap(m=>m.staffEntries).reduce((sum,s)=>sum+s.FingeringEntries.length,0);
  const expectedFingers=(data.readable.musicxml.match(/<fingering\b/g)||[]).length;
  check(fingers>0||expectedFingers===0,'Readable score lost all fingerings');
  await renderScoreView();
  check(notationVersion.readability,'Re-render lost the optimized score');
  check(scoreAnchors.length===new Set(notationVersion.notes.map(n=>n.start)).size,'Optimized-view cursor lost attacks');
  check(JSON.stringify(data.version.notes)===original,'Display changed playback notes');
  $('metronome').checked=true;$('metronome').dispatchEvent(new Event('change'));

  };

  for(const beat of clock.beats)clock.click(beat.seconds,beat.accent);
  const clickSamples=(await clickContext.startRendering()).getChannelData(0),energy=(a,b)=>{
    const part=clickSamples.slice(Math.floor(a*rate),Math.floor(b*rate));return Math.sqrt(part.reduce((s,v)=>s+v*v,0)/part.length);
  };
