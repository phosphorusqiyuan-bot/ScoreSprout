(async()=>{
  const legacy=(await window.scoresprout.listLibrary()).find(item=>item.title==='旧版录音');
  if(!legacy)throw Error('legacy MP3 score is missing');
  await openLibrary(legacy);
  const oldSongId=state.source.id;
  if($('retranscribeMp3').classList.contains('hidden'))throw Error('retranscribe button is hidden');
  await retranscribeMp3();
  const newSongId=state.source.id;
  if(newSongId===oldSongId)throw Error('legacy MP3 was not upgraded');
  const oldScoreKept=(await window.scoresprout.listLibrary()).some(item=>item.songId===oldSongId&&item.id===legacy.id);
  if(!oldScoreKept)throw Error('old score was overwritten');
  return {oldSongId,newSongId,oldScoreKept};
})()
